cmake_minimum_required(VERSION 3.25)
project(Migratool VERSION 2.1.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(NOT WIN32)
  message(FATAL_ERROR "La interfaz y el ejecutor de Migratool requieren Windows.")
endif()
find_program(VCPKG_EXE vcpkg)
if(VCPKG_EXE)
  get_filename_component(VCPKG_DETECTED_DIR "${VCPKG_EXE}" DIRECTORY)
endif()
set(VCPKG_HINTS
  "$ENV{VCPKG_ROOT}/installed/x64-windows"
  "${VCPKG_DETECTED_DIR}/installed/x64-windows"
  "C:/vcpkg/installed/x64-windows"
)
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp HINTS ${VCPKG_HINTS} PATH_SUFFIXES include REQUIRED)
set(CLANG_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/llvm-headers")
find_file(CLANG_DLL libclang.dll
  HINTS "$ENV{LLVM_DIR}/bin" "$ENV{ProgramFiles}/LLVM/bin" "$ENV{LOCALAPPDATA}/Programs/Swift/Toolchains/6.3.2+Asserts/usr/bin"
  PATHS ENV PATH
  REQUIRED)
find_library(SQLITE_LIBRARY sqlite3 HINTS ${VCPKG_HINTS} PATH_SUFFIXES lib REQUIRED)
find_file(SQLITE_DLL sqlite3.dll HINTS ${VCPKG_HINTS} PATH_SUFFIXES bin PATHS ENV PATH REQUIRED)
find_file(ZIP_DLL NAMES libzip.dll zip.dll HINTS "$ENV{MSYS2_ROOT}/ucrt64/bin" "C:/msys64/ucrt64/bin" PATHS ENV PATH REQUIRED)
get_filename_component(ZIP_DLL_DIR "${ZIP_DLL}" DIRECTORY)
get_filename_component(ZIP_PREFIX_DIR "${ZIP_DLL_DIR}" DIRECTORY)
find_path(ZIP_INCLUDE_DIR zip.h HINTS "${ZIP_PREFIX_DIR}/include" "$ENV{MSYS2_ROOT}/ucrt64/include" "C:/msys64/ucrt64/include" ${VCPKG_HINTS} PATH_SUFFIXES include REQUIRED)
foreach(header zip.h zipconf.h)
  configure_file("${ZIP_INCLUDE_DIR}/${header}" "${CMAKE_CURRENT_BINARY_DIR}/zip-headers/${header}" COPYONLY)
endforeach()
find_program(DOTNET dotnet REQUIRED)
file(GLOB_RECURSE IDENTITY_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/include/*.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/adapters/*.cs" "${CMAKE_CURRENT_SOURCE_DIR}/adapters/*.csproj" "${CMAKE_CURRENT_SOURCE_DIR}/resources/*")
list(APPEND IDENTITY_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt")
list(FILTER IDENTITY_SOURCES EXCLUDE REGEX "/(obj|bin)/")
list(SORT IDENTITY_SOURCES)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${IDENTITY_SOURCES})
set(IDENTITY_TEXT "${PROJECT_VERSION}")
foreach(source IN LISTS IDENTITY_SOURCES)
  file(SHA256 "${source}" source_hash)
  file(RELATIVE_PATH source_name "${CMAKE_CURRENT_SOURCE_DIR}" "${source}")
  string(APPEND IDENTITY_TEXT "${source_name}:${source_hash}\n")
endforeach()
string(SHA256 SOURCE_IDENTITY "${IDENTITY_TEXT}")
string(SUBSTRING "${SOURCE_IDENTITY}" 0 16 SHORT_IDENTITY)
add_library(migratool_core src/platform.cpp src/analyzers.cpp src/analysis_context.cpp src/audit.cpp src/database.cpp src/indexer.cpp src/gateway.cpp src/mcp.cpp src/watcher.cpp src/live.cpp src/duplicates.cpp src/queries.cpp src/scale.cpp src/scale_evidence.cpp src/scale_export.cpp src/usn.cpp src/reuse.cpp src/fs_index.cpp src/fs_crawler.cpp src/fs_organize.cpp src/memories.cpp src/file_toolkit.cpp src/workspace_registry.cpp src/lsp_client.cpp src/housekeep.cpp)
target_include_directories(migratool_core PUBLIC include "${JSON_INCLUDE_DIR}" PRIVATE "${CLANG_INCLUDE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/zip-headers")
target_compile_definitions(migratool_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
set_property(SOURCE src/scale.cpp APPEND PROPERTY COMPILE_DEFINITIONS MIGRATOOL_VERSION="${PROJECT_VERSION}" MIGRATOOL_BUILD_ID="${PROJECT_VERSION}-${SHORT_IDENTITY}" MIGRATOOL_SOURCE_SHA256="${SOURCE_IDENTITY}")
target_compile_definitions(migratool_core PRIVATE MIGRATOOL_CLANG_DLL=L"${CLANG_DLL}")
target_compile_definitions(migratool_core PRIVATE MIGRATOOL_ZIP_DLL=L"${ZIP_DLL}")
target_compile_definitions(migratool_core PRIVATE MIGRATOOL_REGISTRY_DEFAULT=L"${CMAKE_CURRENT_SOURCE_DIR}/.migratool/projects.sqlite")
target_link_libraries(migratool_core PUBLIC bcrypt ole32 shell32 shlwapi advapi32 xmllite psapi "${SQLITE_LIBRARY}")
if(MSVC)
  target_compile_options(migratool_core PUBLIC /utf-8 /W4 /permissive- /EHsc)
endif()
add_executable(migratool src/main.cpp resources/app.manifest)
set_target_properties(migratool PROPERTIES OUTPUT_NAME "migratool-cli-0.3")
target_link_libraries(migratool PRIVATE migratool_core)
add_executable(Migratool WIN32 src/gui.cpp src/comparison_view.cpp src/scale_view.cpp resources/app.manifest)
add_executable(census_crosscheck tests/census_crosscheck.cpp resources/app.manifest)
target_link_libraries(census_crosscheck PRIVATE migratool_core)
set_target_properties(Migratool PROPERTIES OUTPUT_NAME "Migratool-0.3")
target_include_directories(Migratool PRIVATE resources)
target_link_libraries(Migratool PRIVATE migratool_core comctl32 comdlg32 uxtheme dwmapi oleacc uuid windowscodecs)
if(MSVC)
  target_compile_options(Migratool PRIVATE /Zi)
  target_link_options(Migratool PRIVATE /DEBUG /OPT:REF /OPT:ICF)
endif()
add_custom_target(roslyn_adapter ALL
  COMMAND "${DOTNET}" build "${CMAKE_CURRENT_SOURCE_DIR}/adapters/roslyn/RoslynAdapter.csproj" -c Release -o "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/adapters/roslyn" --nologo --configfile "${CMAKE_CURRENT_SOURCE_DIR}/NuGet.Config"
  WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" VERBATIM)
add_dependencies(migratool roslyn_adapter)
add_dependencies(Migratool roslyn_adapter)
foreach(app migratool Migratool)
  add_custom_command(TARGET ${app} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SQLITE_DLL}" "$<TARGET_FILE_DIR:${app}>/sqlite3.dll")
endforeach()
# Keep canonical unversioned alias migratool-cli.exe synchronized (tolerant if a live MCP session holds a lock)

  COMMAND cmd /c "copy /Y \"$<TARGET_FILE:migratool>\" \"$<TARGET_FILE_DIR:migratool>\\migratool-cli.exe\" >nul 2>nul || exit /b 0")
